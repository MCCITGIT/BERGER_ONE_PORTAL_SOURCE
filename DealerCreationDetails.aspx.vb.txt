    Private Sub SetDropdownValueSafe(ByVal ddl As DropDownList, ByVal rawValue As String)
        If ddl Is Nothing Then Exit Sub

        Dim targetValue As String = If(rawValue, String.Empty).Trim()
        ddl.ClearSelection()

        If targetValue = String.Empty Then Exit Sub

        Dim matchedItem As ListItem = ddl.Items.FindByValue(targetValue)
        Dim normalizedTarget As String = NormalizeForCompare(targetValue)
        Dim candidates As New List(Of String) From {targetValue}

        For Each token As String In targetValue.Split(New Char() {"-"c, "|"c, ":"c, ","c, "/"c, "("c, ")"c}, StringSplitOptions.RemoveEmptyEntries)
            Dim cleanToken As String = token.Trim()
            If cleanToken <> String.Empty AndAlso Not candidates.Exists(Function(x) x.Equals(cleanToken, StringComparison.OrdinalIgnoreCase)) Then
                candidates.Add(cleanToken)
            End If
        Next

        If matchedItem Is Nothing Then
            For Each item As ListItem In ddl.Items
                Dim itemValue As String = item.Value.Trim()
                Dim itemText As String = item.Text.Trim()
                Dim normalizedItemValue As String = NormalizeForCompare(itemValue)
                Dim normalizedItemText As String = NormalizeForCompare(itemText)

                If candidates.Exists(Function(x) itemValue.Equals(x, StringComparison.OrdinalIgnoreCase) OrElse itemText.Equals(x, StringComparison.OrdinalIgnoreCase)) OrElse
                   (normalizedTarget <> String.Empty AndAlso (normalizedItemValue = normalizedTarget OrElse normalizedItemText = normalizedTarget)) OrElse
                   (normalizedTarget <> String.Empty AndAlso (normalizedItemValue.Contains(normalizedTarget) OrElse normalizedItemText.Contains(normalizedTarget) OrElse normalizedTarget.Contains(normalizedItemValue) OrElse normalizedTarget.Contains(normalizedItemText))) Then
                    matchedItem = item
                    Exit For
                End If
            Next
        End If

        If matchedItem IsNot Nothing Then
            matchedItem.Selected = True
        End If
    End Sub

    Private Function NormalizeForCompare(ByVal input As String) As String
        If String.IsNullOrWhiteSpace(input) Then Return String.Empty

        Dim sb As New System.Text.StringBuilder()
        For Each ch As Char In input.Trim().ToUpperInvariant()
            If Char.IsLetterOrDigit(ch) Then
                sb.Append(ch)
            End If
        Next

        Return sb.ToString()
    End Function

    Private Sub SetTextBoxesReadOnlyRecursive(ByVal parentControl As Control)
        For Each childControl As Control In parentControl.Controls
            If TypeOf childControl Is TextBox Then
                Dim txtControl As TextBox = CType(childControl, TextBox)
                Dim textBoxId As String = If(txtControl.ID, String.Empty)

                If textBoxId.Equals("txtRejectReason", StringComparison.OrdinalIgnoreCase) Then
                    txtControl.Attributes.Remove("readonly")
                Else
                    txtControl.Attributes("readonly") = "readonly"
                End If
            End If

            If childControl.HasControls() Then
                SetTextBoxesReadOnlyRecursive(childControl)
            End If
        Next
    End Sub

#Region "Date Format"
    Public Function FormatDate(ByVal stringdate As String) As String
        Dim dt As String = String.Empty
        If (stringdate.Equals(String.Empty)) Then
            Return String.Empty
        End If

        If Not (stringdate = String.Empty) Then
            Dim ddate As String() = stringdate.Split("/")
            Dim arrlist As New ArrayList
            Dim index As Integer = 0

            While index <= ddate.Length - 1
                arrlist.Add(ddate(index))
                System.Math.Min(System.Threading.Interlocked.Increment(index), index - 1)
            End While
            Dim dd As String = System.Convert.ToString(arrlist.Item(0))
            Dim mm As String = System.Convert.ToString(arrlist.Item(1))
            Dim yyyy As String = System.Convert.ToString(arrlist.Item(2))

            'Dim dt As DateTime = New DateTime(yyyy, mm, dd)
            'dt = FormatDateTime(dt, Microsoft.VisualBasic.DateFormat.LongDate)

            dt = yyyy.ToString + "-" + mm.ToString + "-" + dd.ToString

        End If
        Return dt
    End Function
#End Region
#Region "Date Format GST"
    Public Function FormatDateGST(ByVal stringdate As String) As String
        Dim dt As String = String.Empty
        If (stringdate.Equals(String.Empty)) Then
            dt = SqlDateTime.MinValue.ToString("dd-MM-yyyy")
        End If

        If Not (stringdate = String.Empty) Then
            Dim ddate As String() = stringdate.Split("-")
            Dim arrlist As New ArrayList
            Dim index As Integer = 0

            While index <= ddate.Length - 1
                arrlist.Add(ddate(index))
                System.Math.Min(System.Threading.Interlocked.Increment(index), index - 1)
            End While
            Dim dd As String = System.Convert.ToString(arrlist.Item(2))
            Dim mm As String = System.Convert.ToString(arrlist.Item(1))
            Dim yyyy As String = System.Convert.ToString(arrlist.Item(0))

            'Dim dt As DateTime = New DateTime(yyyy, mm, dd)
            'dt = FormatDateTime(dt, Microsoft.VisualBasic.DateFormat.LongDate)
            dt = dd.ToString + "/" + mm.ToString + "/" + yyyy.ToString


        End If
        Return dt
    End Function

    Private Sub ToggleShipToByCustomerType()

        Dim custType As String = ddlCustomerType.SelectedValue.Trim()


        Dim disableShipToTypes As String() = {"58", "51", "20", "49", "61"}

        If disableShipToTypes.Contains(custType) Then
            EnableShipPanelIfAllowed()
        Else
            DisableShipPanel()
        End If
        If ddlGSTINAvailable.SelectedValue = "Y" Then
            ddlState.Enabled = False
            ddlShipState.Enabled = False
        Else
            ddlState.Enabled = True
        End If

        If ddlGSTINAvailable.SelectedValue = "N" Then
            ' Validation: Bill To State and Ship To State should be same
            If ddlState.SelectedValue <> ddlShipState.SelectedValue Then
                lblValidationMessage.Text = "Bill To State and Ship To State should be the same when GSTIN is not available."
                Exit Sub
            End If
        End If

    End Sub

    Public Function EntityToDataTable(Of T)(ByVal entity As T) As DataTable
        Dim dt As New DataTable(GetType(T).Name)

        ' Get all public properties of the entity
        Dim props = GetType(T).GetProperties()

        ' Add columns
        For Each prop In props
            Dim colType As Type = If(prop.PropertyType.IsGenericType AndAlso
                             prop.PropertyType.GetGenericTypeDefinition() = GetType(Nullable(Of )),
                             Nullable.GetUnderlyingType(prop.PropertyType),
                             prop.PropertyType)

            dt.Columns.Add(prop.Name, colType)
        Next

        ' Add row with values
        Dim row = dt.NewRow()
        For Each prop In props
            row(prop.Name) = If(prop.GetValue(entity, Nothing), DBNull.Value)
        Next
        dt.Rows.Add(row)

        Return dt
    End Function
    Public Function ListToDataTable(Of T)(ByVal entities As IEnumerable(Of T)) As DataTable
        Dim dt As New DataTable(GetType(T).Name)

        ' Get all public properties of T
        Dim props = GetType(T).GetProperties()

        ' Add columns
        For Each prop In props
            Dim colType As Type = If(prop.PropertyType.IsGenericType AndAlso
                         prop.PropertyType.GetGenericTypeDefinition() = GetType(Nullable(Of )),
                         Nullable.GetUnderlyingType(prop.PropertyType),
                         prop.PropertyType)

            dt.Columns.Add(prop.Name, colType)
        Next

        ' Add rows for each entity in the list
        For Each entity In entities
            Dim row = dt.NewRow()
            For Each prop In props
                row(prop.Name) = If(prop.GetValue(entity, Nothing), DBNull.Value)
            Next
            dt.Rows.Add(row)
        Next

        Return dt
    End Function

    Private Function IsValidPAN(panNumber As String) As Boolean
        Dim panRegex As New System.Text.RegularExpressions.Regex("^[A-Z]{5}[0-9]{4}[A-Z]{1}$")
        Return panRegex.IsMatch(panNumber)
    End Function

    Private Function SaveDocument(ByRef fileUploadDoc As System.Web.UI.WebControls.FileUpload) As String
        Try
            If fileUploadDoc.PostedFile Is Nothing OrElse fileUploadDoc.PostedFile.ContentLength <= 0 Then
                Return String.Empty
            End If

            Dim extension As String = System.IO.Path.GetExtension(fileUploadDoc.FileName)
            Dim sourcePath As String = ConfigurationManager.AppSettings.Get("UPLOAD_DOCS_FOLDER_ABS_PATH")
            Dim projectPath As String = sourcePath & "\" & "DEALER_CREATION_DOC"
            Dim fileName As String = Guid.NewGuid().ToString() & extension
            Dim saveLocation As String = projectPath & "\" & fileName

            If Not Directory.Exists(projectPath) Then
                Directory.CreateDirectory(projectPath)
            End If

            fileUploadDoc.PostedFile.SaveAs(saveLocation)

            Return "DEALER_CREATION_DOC/" & fileName
        Catch ex As Exception
            Throw ex
        End Try
    End Function

    'Set this to True when you want to activate new document matrix validation.
    Private ReadOnly UseNewDocumentGridValidation As Boolean = True

    Private Function ValidateDocumentGrid(customerType As String,
                                          customerClass As String,
                                          customerSubType As String,
                                          documentGridData As DataTable,
                                          ByRef validationMessage As String) As Boolean
        validationMessage = String.Empty

        If documentGridData Is Nothing OrElse documentGridData.Rows.Count = 0 Then
            validationMessage = "Please enter atleast one record..!!"
            Return False
        End If

        Dim customerTypeCode As String = If(customerType, String.Empty).Trim()
        Dim customerClassValue As String = If(customerClass, String.Empty).Trim()
        Dim customerSubTypeValue As String = If(customerSubType, String.Empty).Trim()
        Dim relationCode As String = ResolveMotherChildCode(customerTypeCode, customerClassValue, customerSubTypeValue)
        Dim matrixKey As String = GetDocumentMatrixKey(customerTypeCode, relationCode)
        Dim gstFlag As String = If(ddlGSTINAvailable.SelectedValue, String.Empty).Trim().ToUpperInvariant()
        Dim securityChequeFlag As String = If(ddlSecurityCheque.SelectedValue, String.Empty).Trim().ToUpperInvariant()

        Dim uploadedDocs As New HashSet(Of String)(StringComparer.OrdinalIgnoreCase)
        For Each row As DataRow In documentGridData.Rows
            If row IsNot Nothing AndAlso documentGridData.Columns.Contains("DocType") Then
                Dim docType As String = If(Convert.ToString(row("DocType")), String.Empty).Trim()
                If Not String.IsNullOrEmpty(docType) Then
                    uploadedDocs.Add(docType)
                End If
            End If
        Next

        Dim requiredDocs As New List(Of String)()
        requiredDocs.Add("PAN Card of Business/Owner")
        requiredDocs.Add("Aadhar Card")
        requiredDocs.Add("Shop Image")
        requiredDocs.Add("Dealer Image")
        requiredDocs.Add("Signed Dealer Opening Form")

        Select Case matrixKey
            Case "58_MOTHER", "53_MOTHER", "51_MOTHER"
                requiredDocs.Add("MOU")

            Case "58_CHILD", "53_CHILD", "51_CHILD", "61_CHILD", "49_CHILD"
                requiredDocs.Add("Declaration Letter for Retailer Onboarding")
        End Select

        Select Case matrixKey
            Case "58_MOTHER", "53_MOTHER", "53_CHILD", "51_MOTHER", "61_MOTHER", "49_MOTHER", "5", "6", "66", "20", "12", "7"
                If securityChequeFlag = "Y" Then
                    requiredDocs.Add("Security Cheque")
                End If
                'requiredDocs.Add("Trade Licence")

                If gstFlag = "N" Then
                    requiredDocs.Add("GST Unregistered declaration")
                End If
        End Select

        If gstFlag = "Y" Then
            Dim gstCertificateNotRequiredKeys As String() = {"58_CHILD", "51_CHILD", "61_CHILD", "49_CHILD"}
            If Array.IndexOf(gstCertificateNotRequiredKeys, matrixKey) = -1 Then
                requiredDocs.Add("GST Certificate")
            End If
        End If

        Dim missingDocs As New List(Of String)()
        For Each requiredDoc As String In requiredDocs
            If Not IsDocumentUploaded(uploadedDocs, requiredDoc) Then
                missingDocs.Add(requiredDoc)
            End If
        Next

        If missingDocs.Count > 0 Then
            validationMessage = "Please upload required document(s): " & String.Join(", ", missingDocs.ToArray())
            Return False
        End If

        Return True
    End Function

    Private Function ResolveMotherChildCode(customerType As String,
                                            customerClass As String,
                                            customerSubType As String) As String
        Dim custType As String = If(customerType, String.Empty).Trim()
        Dim custClass As String = If(customerClass, String.Empty).Trim().ToUpperInvariant()
        Dim custSubType As String = If(customerSubType, String.Empty).Trim().ToUpperInvariant()

        Select Case custType
            Case "58", "53"
                If custClass = "DISTRIBUTOR ACCOUNT" Then Return "MOTHER"
                If custClass = "RETAILER ACCOUNT" Then Return "CHILD"

            Case "51"
                If custClass = "WHOLESALER ACCOUNT" Then Return "MOTHER"
                If custClass = "SUB DEALER ACCOUNT" Then Return "CHILD"

            Case "61"
                If custSubType = "EXISTING DEALER STORE" Then Return "CHILD"
                If custSubType = "NEW DEALER STORE" Then Return "MOTHER"

            Case "49"
                If custClass = "RETAILER ACCOUNT" Then Return "CHILD"
                Return "MOTHER"
        End Select

        Return "NA"
    End Function

    Private Function GetDocumentMatrixKey(customerType As String, relationCode As String) As String
        Dim custType As String = If(customerType, String.Empty).Trim()
        Dim relation As String = If(relationCode, "NA").Trim().ToUpperInvariant()

        If custType = "58" OrElse custType = "53" OrElse custType = "51" OrElse custType = "61" OrElse custType = "49" Then
            Return custType & "_" & relation
        End If

        Return custType
    End Function

    Private Function IsDocumentUploaded(uploadedDocs As HashSet(Of String), requiredDoc As String) As Boolean
        If uploadedDocs Is Nothing OrElse String.IsNullOrWhiteSpace(requiredDoc) Then
            Return False
        End If

        Select Case requiredDoc.Trim().ToUpperInvariant()
            Case "DECLARATION LETTER FOR RETAILER ONBOARDING"
                Return uploadedDocs.Contains("Declaration Letter for Retailer Onboarding") OrElse uploadedDocs.Contains("Declaration Letter")
            Case "GST UNREGISTERED DECLARATION"
                Return uploadedDocs.Contains("GST Unregistered declaration") OrElse uploadedDocs.Contains("GSTN unregistered declaration form")
            Case "DEALER IMAGE"
                Return uploadedDocs.Contains("Dealer Image") OrElse uploadedDocs.Contains("Deler Image")
            Case Else
                Return uploadedDocs.Contains(requiredDoc)
        End Select
    End Function

#Region "Retailer Opening Restriction Validation Helper"
    Private Function ValidateRetailerOpeningRestriction(customerClass As String,
                                                        customerType As String,
                                                        customerSubTypeValue As String,
                                                        customerSubTypeText As String,
                                                        clubClass As String,
                                                        ByRef validationMessage As String) As Boolean
        validationMessage = String.Empty

        Dim classCode As String = NormalizeRuleValue(customerClass)
        Dim typeCode As String = NormalizeRuleValue(customerType)
        Dim subTypeValueCode As String = NormalizeRuleValue(customerSubTypeValue)
        Dim subTypeTextCode As String = NormalizeRuleValue(customerSubTypeText)
        Dim clubClassCode As String = NormalizeRuleValue(clubClass)

        If classCode = "RETAILER ACCOUNT" AndAlso typeCode = "58" Then
            If Not IsAllowedSubType(subTypeValueCode, subTypeTextCode, "58-RETAILER", "58-REATILER", "BCG-CB") Then
                validationMessage = "For Retailer Account with Customer Type 58, Customer Sub Type must be 58-Retailer or BCG-CB."
                Return False
            End If
        End If

        If classCode = "RETAILER ACCOUNT" AndAlso typeCode = "53" Then
            If IsAllowedSubType(subTypeValueCode, subTypeTextCode, "58-RETAILER", "58-REATILER") Then
                If clubClassCode <> "FUSION" Then
                    validationMessage = "For Retailer Account with Customer Type 53 and Customer Sub Type 58-Retailer, Club Class must be FUSION."
                    Return False
                End If
            ElseIf Not IsAllowedSubType(subTypeValueCode, subTypeTextCode, "53-RETAILER") Then
                validationMessage = "For Retailer Account with Customer Type 53, Customer Sub Type must be 53-Retailer."
                Return False
            End If
        End If

        If classCode = "SUB DEALER ACCOUNT" AndAlso typeCode = "51" Then
            If Not IsAllowedSubType(subTypeValueCode, subTypeTextCode, "NOT REQUIRED") Then
                validationMessage = "For Sub Dealer Account with Customer Type 51, Customer Sub Type must be Not Required."
                Return False
            End If
        End If

        Return True
    End Function

    Private Function IsAllowedSubType(subTypeValueCode As String,
                                      subTypeTextCode As String,
                                      ParamArray allowedSubTypes() As String) As Boolean
        If allowedSubTypes Is Nothing OrElse allowedSubTypes.Length = 0 Then
            Return False
        End If

        For Each allowed As String In allowedSubTypes
            Dim allowedCode As String = NormalizeRuleValue(allowed)
            If subTypeValueCode = allowedCode OrElse subTypeTextCode = allowedCode Then
                Return True
            End If
        Next

        Return False
    End Function

    Private Function NormalizeRuleValue(value As String) As String
        Return If(value, String.Empty).Trim().ToUpperInvariant()
    End Function

    Private Function IsValidEmailAddress(email As String) As Boolean
        If String.IsNullOrWhiteSpace(email) Then
            Return False
        End If

        Try
            Dim mailAddress As New System.Net.Mail.MailAddress(email.Trim())
            Return String.Equals(mailAddress.Address, email.Trim(), StringComparison.OrdinalIgnoreCase)
        Catch
            Return False
        End Try
    End Function

    Protected Sub RejectRequest(ByVal reason As String)
        Dim ul As New DealerCreationClass
        Dim sqlconn As SqlConnection = Nothing
        Dim sqlTrans As SqlTransaction = Nothing
        Try

            If String.IsNullOrEmpty(reason) Then
                lblPopMessage.Text = "Please enter reject reason..!!"
                lblPopMessage.ForeColor = System.Drawing.Color.Green
                ModalPopupExtender1.Show()
                Exit Sub
            End If
            ' Handle approve logic
            Dim entity = New DealerEntityApproval()
            entity.dcmId = Convert.ToInt64(hdnRequestID.Value)
            entity.user_id = userInfo.userIDEntity
            entity.cancel_reason = reason
            entity.approve_yn = "R"

            ul = New DealerCreationClass()
            sqlconn = DBFactory.GetHelper.OpenConnection()
            sqlTrans = sqlconn.BeginTransaction()

            'Dim returnMsg As Integer = ul.DealerCreationApprovalUpdate(entity, sqlconn, sqlTrans)
            Dim returnMsg As Integer = 0

            If (Not (Request.QueryString("ishosales") Is Nothing)) AndAlso (Request.QueryString("ishosales") = "Y") Then
                returnMsg = ul.DealerCreationHoSalesApprovalUpdate(entity, sqlconn, sqlTrans)
            Else
                returnMsg = ul.DealerCreationApprovalUpdate(entity, sqlconn, sqlTrans)
            End If

            If returnMsg > 0 Then
                sqlTrans.Commit()
                lblPopMessage.Text = "Request has been rejected successfully..!!"
                lblPopMessage.ForeColor = System.Drawing.Color.Green
                btnOk.OnClientClick = "return RedirectToListScreen();"
            Else
                sqlTrans.Rollback()
                lblPopMessage.Text = "Error rejecting request."
                lblPopMessage.ForeColor = System.Drawing.Color.Red
                btnOk.OnClientClick = ""
            End If
            ModalPopupExtender1.Show()
        Catch ex As Exception
            If (sqlTrans IsNot Nothing) Then
                sqlTrans.Rollback()
            End If
            lblPopMessage.Text = "Error occurred during rejection."
            lblPopMessage.ForeColor = System.Drawing.Color.Red
            btnOk.OnClientClick = ""
            ModalPopupExtender1.Show()
        Finally
            If (sqlconn IsNot Nothing) Then
                sqlconn.Close()
                sqlconn.Dispose()
            End If
        End Try
    End Sub

    Protected Sub BackToAdminRequest(ByVal reason As String)
        Dim ul As New DealerCreationClass
        Dim sqlconn As SqlConnection = Nothing
        Dim sqlTrans As SqlTransaction = Nothing
        Try

            If String.IsNullOrEmpty(reason) Then
                lblPopMessage.Text = "Please enter reject reason..!!"
                lblPopMessage.ForeColor = System.Drawing.Color.Green
                ModalPopupExtender1.Show()
                Exit Sub
            End If

            ul = New DealerCreationClass()
            sqlconn = DBFactory.GetHelper.OpenConnection()
            sqlTrans = sqlconn.BeginTransaction()

            Dim rowsAffected As Integer = ul.DealerCreationBackToAdminUpdate(Convert.ToInt64(hdnRequestID.Value), userInfo.userIDEntity, reason, sqlconn, sqlTrans)

            If rowsAffected > 0 Then
                sqlTrans.Commit()
                lblPopMessage.Text = "Request has been sent back to Admin successfully."
                lblPopMessage.ForeColor = System.Drawing.Color.Green
                btnOk.OnClientClick = "return RedirectToListScreen();"
            Else
                sqlTrans.Rollback()
                lblPopMessage.Text = Constant.ErrorMessages.GeneralError
                lblPopMessage.ForeColor = System.Drawing.Color.Red
                btnOk.OnClientClick = ""
            End If
            ModalPopupExtender1.Show()
        Catch ex As Exception
            If (sqlTrans IsNot Nothing) Then
                sqlTrans.Rollback()
            End If
            lblPopMessage.Text = Constant.ErrorMessages.GeneralError
            lblPopMessage.ForeColor = System.Drawing.Color.Red
            btnOk.OnClientClick = ""
            ModalPopupExtender1.Show()
        Finally
            If (sqlconn IsNot Nothing) Then
                sqlconn.Close()
                sqlconn.Dispose()
            End If
        End Try
    End Sub

    Protected Sub btnSubmitModal_Click(ByVal sender As Object, ByVal e As EventArgs)
        Dim cancelReason As String = txtRejectReason.Text.Trim()


        If hdnTargetID1.Value = "Approve" Then
            ApproveRequest(cancelReason)
            Return
        End If

        If String.IsNullOrEmpty(cancelReason) Then
            Label4.Text = "Please enter a reason before submitting."
            Label4.ForeColor = System.Drawing.Color.Red
            ModalPopupExtender3.Show()
            Return
        End If

        Label4.Text = "Please Enter Reason"
        Label4.ForeColor = System.Drawing.Color.Red
        ModalPopupExtender3.Hide()

        If hdnTargetID1.Value = "Reject" Then
            RejectRequest(cancelReason)
        ElseIf hdnTargetID1.Value = "BacktoAdmin" Then
            BackToAdminRequest(cancelReason)
        End If
    End Sub

    '==================== Security Cheque UI Rule ====================
    Private Function IsSecurityChequeNotRequired() As Boolean
        Dim subFunc As String = If(ddlSubFunction.SelectedValue, "").Trim().ToUpper()
        Dim custClass As String = If(ddlCustomerClass.SelectedValue, "").Trim().ToUpper()
        Dim custType As String = If(ddlCustomerType.SelectedValue, "").Trim()

        'Rule-1: WholeSaler + Sub Dealer Account + CustomerType 51
        If subFunc.Contains("WHOLESALER") AndAlso custClass.Contains("SUB DEALER ACCOUNT") AndAlso custType = "51" Then
            Return True
        End If

        'Rule-2: Distributor + Retailer Account + CustomerType 58
        If subFunc.Contains("DISTRIBUTOR") AndAlso custClass.Contains("RETAILER ACCOUNT") AndAlso custType = "58" Then
            Return True
        End If

        Return False
    End Function

    Private Sub ApplySecurityChequeRule()
        Dim notRequired As Boolean = IsSecurityChequeNotRequired()

        If notRequired Then
            ddlSecurityCheque.ClearSelection()
            If ddlSecurityCheque.Items.FindByValue("N") IsNot Nothing Then
                ddlSecurityCheque.SelectedValue = "N"
                'txtChequeNumber.Text = String.Empty
            End If

            ddlSecurityCheque.Enabled = False

            pnlSecurityChequeCollected.Visible = False
            pnlSecurityCheque.Visible = False

        Else
            pnlSecurityChequeCollected.Visible = True
            ddlSecurityCheque.Enabled = True

            ddlSecurityCheque_SelectedIndexChanged(Nothing, Nothing)
        End If
    End Sub

    Private Function ValidateAddressAndDeclarationDetails() As Boolean
        If String.IsNullOrEmpty(ddlState.SelectedValue) Then
            Return ShowValidationError("Please select State.")
        End If

        If txtCity.Text.Trim().Length > 50 Then
            Return ShowValidationError("City length must not exceed 50 characters.")
        End If

        If txtShipCity.Text.Trim().Length > 50 Then
            Return ShowValidationError("Ship City length must not exceed 50 characters.")
        End If

        If String.IsNullOrWhiteSpace(ddlAddress1.SelectedValue) Then
            Return ShowValidationError("Please select Address 1.")
        End If

        Dim address2Value As String = txtAddress2.Text.Trim()
        Dim address3Value As String = txtAddress3.Text.Trim()

        If String.IsNullOrWhiteSpace(address2Value) Then
            Return ShowValidationError("Please enter a valid Address 2.")
        End If

        If address2Value.Length < 3 OrElse address2Value.Length > 99 Then
            Return ShowValidationError("Address 2 length must be between 3 and 99 characters.")
        End If

        If String.IsNullOrWhiteSpace(address3Value) Then
            Return ShowValidationError("Please enter a valid Address 3.")
        End If

        If address3Value.Length < 3 OrElse address3Value.Length > 99 Then
            Return ShowValidationError("Address 3 length must be between 3 and 99 characters.")
        End If

        If ddlFinancialTranYn.SelectedValue <> "Y" AndAlso ddlFinancialTranYn.SelectedValue <> "N" Then
            Return ShowValidationError("Please specify details of such transactions.")
        End If

        If ddlFinancialTranYn.SelectedValue = "Y" AndAlso String.IsNullOrWhiteSpace(txtFinancialTranRemark.Text) Then
            Return ShowValidationError("Please Enter Remark.")
        End If

        If ddlRelativeOrFriend.SelectedValue <> "Y" AndAlso ddlRelativeOrFriend.SelectedValue <> "N" Then
            Return ShowValidationError("Please specify details of such relationship.")
        End If

        If ddlRelativeOrFriend.SelectedValue = "Y" AndAlso String.IsNullOrWhiteSpace(txtRelativeOrFriendRemark.Text) Then
            Return ShowValidationError("Please Enter Remark.")
        End If

        Return True
    End Function


    Private Sub Download_Document(ByVal workflowId As String, ByVal emSignerPassword As String)
        ' Ensure TLS 1.2 is enabled
        ServicePointManager.SecurityProtocol = SecurityProtocolType.Tls12

        Dim webClient As New WebClient()

        webClient.Headers(HttpRequestHeader.ContentType) = "application/json"
        webClient.Headers(HttpRequestHeader.Accept) = "application/json"
        webClient.Headers.Add("AppName", "bergerlive")
        webClient.Headers.Add("SecretKey", "d57c7037-dd1b-45c4-a0ae-75d2d0562a77")
        webClient.Headers.Add("Authorization", "")

        ' Create the request object
        Dim objentity As New DownloadAuthRequest With {
            .UserName = "ecom@bergerindia.com",
            .Password = emSignerPassword
        }

        ' Convert the object to JSON
        Dim jsonContent As String = JsonConvert.SerializeObject(objentity)

        Try
            ' Send the request
            Dim response As String = webClient.UploadString("https://api.emsigner.com/api/ValidateLogin", "POST", jsonContent)

            ' Parse JSON response to extract AuthToken
            Dim authData As DownloadAuthResponse = JsonConvert.DeserializeObject(Of DownloadAuthResponse)(response)
            Dim authToken As String = authData.Response.AuthToken

            Dim webClient1 As New WebClient()
            webClient1.Headers.Add("UserName", "ecom@bergerindia.com")
            webClient1.Headers(HttpRequestHeader.ContentType) = "application/json"
            webClient1.Headers(HttpRequestHeader.Authorization) = "Basic " & authToken

            Dim objWorkflow As New DownloadWorkFlowRequest With {
                .WorkflowId = workflowId
            }

            Dim jsonContent1 As String = JsonConvert.SerializeObject(objWorkflow)
            Dim response1 As String = webClient1.UploadString("https://api.emsigner.com/api/DownloadWorkflowDocuments", "POST", jsonContent1)
            Dim responseData As DownloadWorkflowResponse = JsonConvert.DeserializeObject(Of DownloadWorkflowResponse)(response1)

            If responseData IsNot Nothing AndAlso responseData.Response.FileList.Count > 0 Then
                Dim downloadedWorkflowId = responseData.Response.WorkflowId
                Dim fileInfo As DownloadDocumentFile = responseData.Response.FileList(0)

                Dim newFileName As String = downloadedWorkflowId & ".pdf"
                Dim folderPath As String = "E:\VIRTUAL_DOCS\emSignerDocs\"

                If Not Directory.Exists(folderPath) Then
                    Directory.CreateDirectory(folderPath)
                End If

                Dim filePath As String = folderPath & newFileName
                Dim base64Data As String = fileInfo.Base64FileData
                Dim fileBytes As Byte() = Convert.FromBase64String(base64Data)

                If File.Exists(filePath) Then
                    File.Delete(filePath)
                End If

                File.WriteAllBytes(filePath, fileBytes)

                Dim Response_data As HttpResponse = HttpContext.Current.Response
                Response_data.Clear()
                Response_data.ContentType = "application/pdf"
                Response_data.AddHeader("Content-Disposition", "attachment; filename=" & newFileName)
                Response_data.WriteFile(filePath)
                Response_data.Flush()
                HttpContext.Current.ApplicationInstance.CompleteRequest()
            End If

        Catch ex As WebException
            Throw
        End Try
    End Sub

    Protected Sub btnDownloadAgreement_Click(ByVal sender As Object, ByVal e As EventArgs) Handles btnDownloadAgreement.Click
        Dim requestIdInt As Integer
        Integer.TryParse(hdnRequestID.Value, requestIdInt)

        Dim statusObj As New Booking_Aggrement_StatusClass
        Dim ds As DataSet = statusObj.Populate_Booking_Aggrement_Status(requestIdInt)

        If (Not (ds Is Nothing) AndAlso ds.Tables.Count > 0 AndAlso Not (ds.Tables(0) Is Nothing) AndAlso ds.Tables(0).Rows.Count > 0) Then
            Dim statusText As String = ds.Tables(0).Rows(0)("status").ToString().Trim()
            Dim workflowId As String = ds.Tables(0).Rows(0)("workflow_id").ToString().Trim()

            If statusText = "Completed" AndAlso workflowId <> String.Empty Then
                Try
                    Dim passPath As String = ConfigurationManager.AppSettings("DOC_EMSIGNER_PASS_PATH")
                    Dim emSignerPassword As String = File.ReadAllText(passPath)
                    Download_Document(workflowId, emSignerPassword)
                Catch ex As Exception
                    lblPopMessage.Text = "Download agreement failed. " & ex.Message
                    lblPopMessage.ForeColor = System.Drawing.Color.Red
                    ModalPopupExtender1.Show()
                End Try
            Else
                lblPopMessage.Text = "First complete the E-Signing Process."
                lblPopMessage.ForeColor = System.Drawing.Color.Red
                ModalPopupExtender1.Show()
            End If
        Else
            lblPopMessage.Text = "First complete the E-Signing Process."
            lblPopMessage.ForeColor = System.Drawing.Color.Red
            ModalPopupExtender1.Show()
        End If
    End Sub
